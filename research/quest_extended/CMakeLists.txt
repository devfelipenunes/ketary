cmake_minimum_required(VERSION 3.21)
project(KetaryExtendedProbe LANGUAGES C CXX)
if(NOT DEFINED QUEST_SOURCE)
  message(FATAL_ERROR "Supply pinned QuEST checkout")
endif()
set(FLOAT_PRECISION 2 CACHE STRING "" FORCE)
set(ENABLE_MULTITHREADING OFF CACHE BOOL "" FORCE)
set(ENABLE_DISTRIBUTION OFF CACHE BOOL "" FORCE)
set(ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(ENABLE_HIP OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${QUEST_SOURCE}" quest-build EXCLUDE_FROM_ALL)
add_library(ketary_quest_extended SHARED bridge.cpp)
target_link_libraries(ketary_quest_extended PRIVATE QuEST::QuEST)
target_compile_features(ketary_quest_extended PRIVATE cxx_std_17)
