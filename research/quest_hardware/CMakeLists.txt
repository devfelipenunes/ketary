cmake_minimum_required(VERSION 3.21)
project(KetaryHardwareProbe LANGUAGES C CXX)
if(NOT DEFINED QUEST_SOURCE)
  message(FATAL_ERROR "Set QUEST_SOURCE to the pinned QuEST checkout")
endif()
option(KETARY_PROBE_MPI "Enable MPI" OFF)
option(KETARY_PROBE_CUDA "Enable CUDA" OFF)
option(KETARY_PROBE_CUQUANTUM "Enable cuQuantum" OFF)
if(KETARY_PROBE_CUQUANTUM AND NOT KETARY_PROBE_CUDA)
  message(FATAL_ERROR "cuQuantum needs CUDA")
endif()
set(FLOAT_PRECISION 2 CACHE STRING "" FORCE)
set(ENABLE_MULTITHREADING ON CACHE BOOL "" FORCE)
set(ENABLE_DISTRIBUTION ${KETARY_PROBE_MPI} CACHE BOOL "" FORCE)
set(ENABLE_CUDA ${KETARY_PROBE_CUDA} CACHE BOOL "" FORCE)
set(ENABLE_CUQUANTUM ${KETARY_PROBE_CUQUANTUM} CACHE BOOL "" FORCE)
set(ENABLE_HIP OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${QUEST_SOURCE}" quest-build EXCLUDE_FROM_ALL)
add_executable(ketary_hardware_probe probe.cpp)
target_link_libraries(ketary_hardware_probe PRIVATE QuEST::QuEST)
target_compile_features(ketary_hardware_probe PRIVATE cxx_std_17)
